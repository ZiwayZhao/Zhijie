def level_from_xp(total_xp: int) -> int:
    level = 1
    while level < 50 and total_xp >= int(100 * ((level + 1) ** 1.5)):
        level += 1
    return level

def apply_daily_streak(profile, activity_date: date):
    if profile.last_activity_date is None:
        profile.current_streak = 1
    elif activity_date == profile.last_activity_date:
        return profile
    elif activity_date == profile.last_activity_date + timedelta(days=1):
        profile.current_streak += 1
    elif activity_date > profile.last_activity_date + timedelta(days=1):
        gap = (activity_date - profile.last_activity_date).days - 1
        if gap > 0 and profile.streak_freeze_count >= gap:
            profile.streak_freeze_count -= gap
            profile.current_streak += 1
        else:
            profile.current_streak = 1
    profile.longest_streak = max(profile.longest_streak, profile.current_streak)
    profile.last_activity_date = activity_date
    return profile
```

聚合 SQL 思路：
```sql
select
  date_trunc('day', reviewed_at) as day,
  count(*) as review_count,
  avg(case when rating >= 3 then 1.0 else 0.0 end) as accuracy,
  sum(coalesce(response_ms, 0)) / 1000 as review_seconds
from review_logs
where user_id = :user_id
  and reviewed_at >= now() - interval '30 days'
group by 1
order by 1;
```

**3. 四阶段考试备考模型**
模型核心是“倒计时驱动的阶段切换 + mastery 驱动的资源分配”。建议阶段：
- `Phase 1 Survey`: `D-30 ~ D-21`，做诊断测验，识别盲点
- `Phase 2 Learn`: `D-20 ~ D-11`，高权重弱模块集中突破
- `Phase 3 Consolidate`: `D-10 ~ D-4`，FSRS 闪卡 + 错题回顾
- `Phase 4 Sprint`: `D-3 ~ D-0`，整卷模拟、计时、复盘

时间分配函数：
```python
def phase_by_days_left(days_left: int) -> str:
    if days_left >= 21: return "survey"
    if days_left >= 11: return "learn"
    if days_left >= 4: return "consolidate"
    return "sprint"

def allocate_minutes(days_left: int, weak_ratio: float) -> dict[str, int]:
    total = 60 if days_left > 14 else 90 if days_left > 5 else 120
    if days_left >= 21:
        return {"survey": int(total*0.7), "review": int(total*0.3)}
    if days_left >= 11:
        return {"learn": int(total*(0.6 + weak_ratio*0.2)), "review": total//4, "quiz": total//6}
    if days_left >= 4:
        return {"flashcards": total//3, "mistakes": total//3, "mixed_quiz": total - 2*(total//3)}
    return {"mock_exam": int(total*0.6), "error_review": int(total*0.25), "flashcards": total - int(total*0.85)}
```

    return level

def apply_daily_streak(profile, activity_date: date):
    if profile.last_activity_date is None:
        profile.current_streak = 1
    elif activity_date == profile.last_activity_date:
        return profile
    elif activity_date == profile.last_activity_date + timedelta(days=1):
        profile.current_streak += 1
    elif activity_date > profile.last_activity_date + timedelta(days=1):
        gap = (activity_date - profile.last_activity_date).days - 1
        if gap > 0 and profile.streak_freeze_count >= gap:
            profile.streak_freeze_count -= gap
            profile.current_streak += 1
        else:
            profile.current_streak = 1
    profile.longest_streak = max(profile.longest_streak, profile.current_streak)
    profile.last_activity_date = activity_date
    return profile
```

**3. 四阶段考试备考模型**
模型核心是“倒计时驱动的阶段切换 + mastery 驱动的资源分配”。建议阶段：
- `Phase 1 Survey`: `D-30 ~ D-21`，做诊断测验，识别盲点
- `Phase 2 Learn`: `D-20 ~ D-11`，高权重弱模块集中突破
- `Phase 3 Consolidate`: `D-10 ~ D-4`，FSRS 闪卡 + 错题回顾
- `Phase 4 Sprint`: `D-3 ~ D-0`，整卷模拟、计时、复盘

时间分配函数：
```python
def phase_by_days_left(days_left: int) -> str:
    if days_left >= 21: return "survey"
    if days_left >= 11: return "learn"
    if days_left >= 4: return "consolidate"
    return "sprint"

def allocate_minutes(days_left: int, weak_ratio: float) -> dict[str, int]:
    total = 60 if days_left > 14 else 90 if days_left > 5 else 120
    if days_left >= 21:
        return {"survey": int(total*0.7), "review": int(total*0.3)}
    if days_left >= 11:
        return {"learn": int(total*(0.6 + weak_ratio*0.2)), "review": total//4, "quiz": total//6}
    if days_left >= 4:
        return {"flashcards": total//3, "mistakes": total//3, "mixed_quiz": total - 2*(total//3)}
    return {"mock_exam": int(total*0.6), "error_review": int(total*0.25), "flashcards": total - int(total*0.85)}
```
