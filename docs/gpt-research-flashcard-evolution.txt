# app/services/evolution/apply_service.py
def apply(db, evo, reviewer_id: str):
    proposal = evo.proposal
    action = proposal["action"]

    if action == "split":
        retire_original_card(db, evo.card_id)
        for n in proposal["new_notes"]:
            note_id = create_note(db, evo.deck_id, n, parent_note_id=evo.note_id)
            create_card(db, evo.deck_id, note_id, user_id=evo.user_id)

    elif action == "retire":
        retire_original_card(db, evo.card_id)

    elif action == "merge":
        retire_original_card(db, evo.card_id)
        mark_note_merged(db, evo.note_id, proposal["merge_target_note_id"])

    elif action == "rewrite":
        update_note_fields(db, evo.note_id, proposal["rewrite_fields"])

    elif action == "add_hint":
        update_note_hint(db, evo.note_id, proposal["hint"])

    elif action == "add_prerequisite":
        for n in proposal["new_notes"]:
            note_id = create_note(db, evo.deck_id, n, parent_note_id=evo.note_id)
            create_card(db, evo.deck_id, note_id, user_id=evo.user_id)

    mark_evolution_applied(db, evo.id, reviewer_id)

**1. 总体架构**
1. 用户提交一次复习结果到 `POST /api/v1/flashcards/review`。
2. 后端更新 `flashcard_cards` 的 FSRS 状态、写入 `review_logs`。
3. 同步执行 `TriggerDetector`，识别 5 类进化触发。
4. 若命中触发，写入 `evolution_logs(status='pending_llm')`，再异步调用 Claude 生成候选方案。
5. Claude 仅返回结构化 JSON，不直接改库。
6. `QualityGuard` 做最小信息原则检查、TF-IDF 去重、场景合法性校验。
7. 通过校验后把建议保存为 `pending_review`。
8. 前端拉取建议列表，显示“原卡 / 建议后卡 / 风险提示 / 去重结果 / 触发原因”。
9. 用户确认后调用 `POST /api/v1/evolutions/{id}/apply`，事务性执行拆分、退役、合并、改写、加提示或补前置卡。
10. 所有动作都保留审计日志，可回滚为“恢复原 note/card”。

**2. 5 种触发规则**
```python
# app/services/evolution/trigger_detector.py
from dataclasses import dataclass
from typing import Literal

TriggerType = Literal[
    "consecutive_again",
    "consecutive_easy",
    "high_lapse_rate",
    "slow_response",
    "cluster_failure_prereq",
]

@dataclass
class EvolutionTrigger:
    type: TriggerType
    card_id: str
    note_id: str
    deck_id: str
    user_id: str
    metrics: dict

def detect_card_triggers(card, review_stats, cluster_stats) -> list[EvolutionTrigger]:
    out = []

    if card.consecutive_again >= 3:
        out.append(EvolutionTrigger(
            type="consecutive_again",
            card_id=card.id,
            note_id=card.note_id,
            deck_id=card.deck_id,
            user_id=card.user_id,
            metrics={"consecutive_again": card.consecutive_again},
        ))

    if card.consecutive_easy >= 5:
        out.append(EvolutionTrigger(
            type="consecutive_easy",
            card_id=card.id,
            note_id=card.note_id,
            deck_id=card.deck_id,
            user_id=card.user_id,
            metrics={"consecutive_easy": card.consecutive_easy},
        ))

    if card.reps > 10 and review_stats.lapse_rate > 0.5:
        out.append(EvolutionTrigger(
            type="high_lapse_rate",
            card_id=card.id,
            note_id=card.note_id,
            deck_id=card.deck_id,
            user_id=card.user_id,
            metrics={"lapse_rate": review_stats.lapse_rate, "reps": card.reps},
        ))

    if card.average_response_ms and card.average_response_ms > 15000:
        out.append(EvolutionTrigger(
            type="slow_response",
            card_id=card.id,
            note_id=card.note_id,
            deck_id=card.deck_id,
            user_id=card.user_id,
            metrics={"average_response_ms": card.average_response_ms},
        ))

    if cluster_stats.failed_cards >= 3 and cluster_stats.failure_rate >= 0.6:
        out.append(EvolutionTrigger(
            type="cluster_failure_prereq",
            card_id=card.id,
            note_id=card.note_id,
            deck_id=card.deck_id,
            user_id=card.user_id,
            metrics={
                "cluster_key": cluster_stats.cluster_key,
                "failed_cards": cluster_stats.failed_cards,
                "failure_rate": cluster_stats.failure_rate,
            },
        ))

    return out
```

建议把“相关卡片集体失败”定义为：
- 同一 `cluster_key` 下，最近 48 小时或最近 20 次复习内
- 至少 3 张不同卡片
- `Again + Hard` 占比 >= 60%
- `cluster_key` 由 `course_id + module_id + concept_tag` 构成

# app/services/evolution/claude_client.py
from anthropic import Anthropic
from app.schemas.evolution import EvolutionProposal

class ClaudeEvolutionClient:
    def __init__(self, api_key: str, model: str):
        self.client = Anthropic(api_key=api_key)
        self.model = model

    def generate(self, system_prompt: str, user_prompt: str) -> EvolutionProposal:
        schema = EvolutionProposal.model_json_schema()
        resp = self.client.messages.create(
            model=self.model,
            max_tokens=1200,
            system=system_prompt,
            messages=[{"role": "user", "content": user_prompt}],
            tools=[{
                "name": "submit_evolution",
                "description": "Return one validated flashcard evolution proposal",
                "input_schema": schema,
            }],
            tool_choice={"type": "tool", "name": "submit_evolution"},
        )

        tool_use = next(block for block in resp.content if block.type == "tool_use")
        return EvolutionProposal.model_validate(tool_use.input)

**7. 质量控制**
`QualityGuard` 至少做 5 层：

1. 最小信息原则
```python
def check_minimum_information(note) -> list[str]:
    issues = []
    if len(note["back"]) > 120:
        issues.append("answer_too_long")
    if any(x in note["back"] for x in ["、", "以及", "并且", "首先", "然后", "最后"]):
        issues.append("possible_multi_fact")
    if note["front"].count("和") >= 2:
        issues.append("question_overloaded")
    return issues
```

2. TF-IDF 去重  
- 语料：`canonical_question + canonical_answer`
- `cosine >= 0.82`：直接拦截
- `0.72 <= cosine < 0.82`：标记 `duplicate_risk=high`，需人工确认
- 可用 `scikit-learn TfidfVectorizer(ngram_range=(1,2))`

3. 事实边界控制  
只允许基于 `source_snapshot` 改写，不允许引入未提供事实；若 `source_snapshot` 不足，直接拒绝生成。

4. 场景合法性  
- `split` 至少 2 张子卡
- `retire` 不能针对仍高 lapse 的卡
- `merge` 必须指定已存在 `merge_target_note_id`
- `add_hint` 不能包含答案全文
- `add_prerequisite` 必须带 `prerequisite_for`

5. 人工确认流程  
状态机：
`pending_llm -> pending_review -> approved/rejected -> applied`
后台只自动生成，不自动应用；真正改库必须用户点击确认。

**8. RPKT：前置知识递归追踪**
这里建议实现为“知识图 + 轻量 BKT 状态”的混合方案。

知识图来源：
- `flashcard_notes.prerequisite_refs`：显式前置依赖
- `cluster_key / tags / source_ref.module_id`：隐式聚类
- 可选从课程大纲抽出的模块树

每个概念维护 `p_know`：
- 初始值可由该概念关联卡片的 FSRS 稳定度映射得到
- 复习正确时按 BKT 更新
- 复习错误时按 BKT 更新
- 深度回溯时按边权衰减

```python
# app/services/evolution/rpkt.py
def bkt_update(p_know: float, correct: bool, slip=0.1, guess=0.2, learn=0.15) -> float:
    if correct:
        post = (p_know * (1 - slip)) / (p_know * (1 - slip) + (1 - p_know) * guess)
    else:
        post = (p_know * slip) / (p_know * slip + (1 - p_know) * (1 - guess))
    return post + (1 - post) * learn

def trace_missing_prereqs(graph, failed_concepts: list[str], mastery: dict[str, float], max_depth=3):
    scores = {}
    queue = [(c, 0, 1.0) for c in failed_concepts]

    while queue:
        concept, depth, weight = queue.pop(0)
        if depth >= max_depth:
            continue
        for prereq, edge_weight in graph.get(concept, []):
            lack = 1 - mastery.get(prereq, 0.5)
            score = lack * weight * edge_weight * (0.85 ** depth)
            scores[prereq] = max(scores.get(prereq, 0), score)
            queue.append((prereq, depth + 1, weight * edge_weight))

    return sorted(scores.items(), key=lambda x: x[1], reverse=True)
```

触发流程：
1. 找出失败簇里的概念节点
2. 回溯 1~3 层前置图
3. 选 `score > 0.45` 且当前没有对应卡片的前置概念
4. 把这些概念喂给 Claude 生成“前置知识卡”

```python
# app/services/evolution/apply_service.py
def apply(db, evo, reviewer_id: str):
    proposal = evo.proposal
    action = proposal["action"]

    if action == "split":
        retire_original_card(db, evo.card_id)
        for n in proposal["new_notes"]:
            note_id = create_note(db, evo.deck_id, n, parent_note_id=evo.note_id)
            create_card(db, evo.deck_id, note_id, user_id=evo.user_id)

    elif action == "retire":
        retire_original_card(db, evo.card_id)

    elif action == "merge":
        retire_original_card(db, evo.card_id)
        mark_note_merged(db, evo.note_id, proposal["merge_target_note_id"])

    elif action == "rewrite":
        update_note_fields(db, evo.note_id, proposal["rewrite_fields"])

    elif action == "add_hint":
        update_note_hint(db, evo.note_id, proposal["hint"])

    elif action == "add_prerequisite":
        for n in proposal["new_notes"]:
            note_id = create_note(db, evo.deck_id, n, parent_note_id=evo.note_id)
            create_card(db, evo.deck_id, note_id, user_id=evo.user_id)

    mark_evolution_applied(db, evo.id, reviewer_id)
```
